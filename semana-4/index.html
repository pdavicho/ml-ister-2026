# ── Parte 1: Matriz de Confusión y Métricas — Dataset Wine ──
from sklearn.datasets        import load_wine
from sklearn.ensemble        import RandomForestClassifier
from sklearn.tree            import DecisionTreeClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics         import (confusion_matrix, classification_report,
                                     accuracy_score, ConfusionMatrixDisplay)
import matplotlib.pyplot as plt

data = load_wine()
X_train, X_test, y_train, y_test = train_test_split(
    data.data, data.target, test_size=0.30, random_state=42)

rf = RandomForestClassifier(n_estimators=100, random_state=42, n_jobs=-1)
rf.fit(X_train, y_train)
y_pred = rf.predict(X_test)

# Matriz de confusión visual
fig, ax = plt.subplots(figsize=(6, 5))
ConfusionMatrixDisplay.from_predictions(
    y_test, y_pred,
    display_labels=data.target_names,
    colorbar=False,
    ax=ax
)
ax.set_title('Matriz de Confusión — Random Forest en Wine')
plt.tight_layout()
plt.show()

print("\n── Classification Report ────────────────────────────")
print(classification_report(y_test, y_pred, target_names=data.target_names))
# ── Parte 2: Cross-Validation — Árbol vs RF ───────────────
from sklearn.model_selection import cross_val_score
import numpy as np

arbol = DecisionTreeClassifier(max_depth=5, random_state=42)
rf100 = RandomForestClassifier(n_estimators=100, random_state=42, n_jobs=-1)

cv_arbol = cross_val_score(arbol, data.data, data.target, cv=10, scoring='accuracy')
cv_rf    = cross_val_score(rf100, data.data, data.target, cv=10, scoring='accuracy')

print("── Comparativa CV-10 — Dataset Wine ─────────────────")
print(f"Árbol  | Media: {cv_arbol.mean():.4f} | Std: {cv_arbol.std():.4f} | Min: {cv_arbol.min():.4f} | Max: {cv_arbol.max():.4f}")
print(f"RF 100 | Media: {cv_rf.mean():.4f} | Std: {cv_rf.std():.4f} | Min: {cv_rf.min():.4f} | Max: {cv_rf.max():.4f}")

# Gráfico de barras de los 10 folds
x = np.arange(1, 11)
plt.figure(figsize=(9, 4))
plt.bar(x - 0.2, cv_arbol, width=0.4, label=f'Árbol (μ={cv_arbol.mean():.3f})', color='#f59e0b', alpha=0.8)
plt.bar(x + 0.2, cv_rf,    width=0.4, label=f'RF 100 (μ={cv_rf.mean():.3f})',   color='#3b82f6', alpha=0.8)
plt.axhline(cv_arbol.mean(), color='#f59e0b', linestyle='--', linewidth=1.2)
plt.axhline(cv_rf.mean(),    color='#3b82f6',  linestyle='--', linewidth=1.2)
plt.xlabel('Fold')
plt.ylabel('Accuracy')
plt.title('Cross-Validation 10-Fold — Árbol vs RF en Wine')
plt.legend()
plt.ylim(0.7, 1.05)
plt.grid(axis='y', alpha=0.3)
plt.tight_layout()
plt.show()
# ── Parte 3: Curva ROC y AUC — Breast Cancer (binario) ────
from sklearn.datasets import load_breast_cancer
from sklearn.metrics  import roc_curve, roc_auc_score

bc = load_breast_cancer()
X_tr, X_te, y_tr, y_te = train_test_split(
    bc.data, bc.target, test_size=0.30, random_state=42)

arbol_bc = DecisionTreeClassifier(max_depth=5, random_state=42)
rf_bc    = RandomForestClassifier(n_estimators=100, random_state=42, n_jobs=-1)
arbol_bc.fit(X_tr, y_tr)
rf_bc.fit(X_tr, y_tr)

prob_arbol = arbol_bc.predict_proba(X_te)[:, 1]
prob_rf    = rf_bc.predict_proba(X_te)[:, 1]

fpr_a, tpr_a, _ = roc_curve(y_te, prob_arbol)
fpr_r, tpr_r, _ = roc_curve(y_te, prob_rf)

auc_a = roc_auc_score(y_te, prob_arbol)
auc_r = roc_auc_score(y_te, prob_rf)

plt.figure(figsize=(6, 5))
plt.plot(fpr_a, tpr_a, color='#f59e0b', lw=2, label=f'Árbol (AUC = {auc_a:.3f})')
plt.plot(fpr_r, tpr_r, color='#3b82f6', lw=2, label=f'RF 100 (AUC = {auc_r:.3f})')
plt.plot([0, 1], [0, 1], 'k--', lw=1, label='Aleatorio (AUC = 0.50)')
plt.xlabel('False Positive Rate (1 - Especificidad)')
plt.ylabel('True Positive Rate (Sensibilidad)')
plt.title('Curva ROC — Árbol vs RF en Breast Cancer')
plt.legend(loc='lower right')
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()
# ── Parte 4: Curva de Aprendizaje — diagnosticar overfitting
from sklearn.model_selection import learning_curve

train_sizes, train_scores, val_scores = learning_curve(
    RandomForestClassifier(n_estimators=100, random_state=42, n_jobs=-1),
    data.data, data.target,
    train_sizes=np.linspace(0.1, 1.0, 10),
    cv=5, scoring='accuracy', n_jobs=-1
)

train_mean = train_scores.mean(axis=1)
train_std  = train_scores.std(axis=1)
val_mean   = val_scores.mean(axis=1)
val_std    = val_scores.std(axis=1)

plt.figure(figsize=(8, 4))
plt.plot(train_sizes, train_mean, 'o-', color='#f59e0b', lw=2, label='Train accuracy')
plt.fill_between(train_sizes, train_mean - train_std, train_mean + train_std, alpha=0.15, color='#f59e0b')
plt.plot(train_sizes, val_mean, 'o-', color='#3b82f6', lw=2, label='Validación (CV-5)')
plt.fill_between(train_sizes, val_mean - val_std, val_mean + val_std, alpha=0.15, color='#3b82f6')
plt.xlabel('Ejemplos de entrenamiento')
plt.ylabel('Accuracy')
plt.title('Curva de Aprendizaje — Random Forest en Wine')
plt.legend()
plt.grid(alpha=0.3)
plt.ylim(0.7, 1.05)
plt.tight_layout()
plt.show()

brecha = train_mean[-1] - val_mean[-1]
print(f"Train acc final:    {train_mean[-1]:.4f}")
print(f"Val acc final:      {val_mean[-1]:.4f}")
print(f"Brecha (overfitting? si > 0.05): {brecha:.4f}")
