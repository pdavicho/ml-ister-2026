# ── Parte 1: Cargar y Explorar el Dataset Auto MPG ──────────
import numpy as np
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# Auto MPG: características de automóviles clásicos (sin etiqueta objetivo)
df = sns.load_dataset('mpg').dropna().reset_index(drop=True)
print(f"Autos: {df.shape[0]}  |  Columnas: {list(df.columns)}")

# Features numéricas para agrupar (NO usamos 'origin' ni 'name' como entrada)
features = ['mpg', 'displacement', 'horsepower', 'weight', 'acceleration']
X = df[features].copy()
print(X.describe().round(1))

# Nota: las escalas son MUY distintas (weight ~3000, mpg ~25) → habrá que escalar
# ── Parte 2: Escalar y Aplicar K-means (k=3) ────────────────
from sklearn.preprocessing import StandardScaler
from sklearn.cluster       import KMeans

# 1) Escalado: imprescindible antes de medir distancias
scaler  = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 2) K-means con 3 clústeres
kmeans = KMeans(n_clusters=3, init='k-means++', n_init=10, random_state=42)
clusters = kmeans.fit_predict(X_scaled)
df['cluster'] = clusters

print("Tamaño de cada clúster:")
print(pd.Series(clusters).value_counts().sort_index())
print(f"\nInercia (compacidad total): {kmeans.inertia_:.1f}")

# Visualizar en 2 dimensiones intuitivas: peso vs consumo
plt.figure(figsize=(8, 5))
plt.scatter(df['weight'], df['mpg'], c=clusters, cmap='viridis', s=35, edgecolor='k', linewidth=0.3)
plt.xlabel('Peso (weight)')
plt.ylabel('Consumo (mpg)')
plt.title('K-means (k=3) sobre automóviles — Peso vs Consumo')
plt.colorbar(label='Clúster')
plt.tight_layout()
plt.show()
# ── Parte 3: ¿Cuántos clústeres? Método del Codo ────────────
inercias = []
ks = range(1, 11)
for k in ks:
    km = KMeans(n_clusters=k, init='k-means++', n_init=10, random_state=42)
    km.fit(X_scaled)
    inercias.append(km.inertia_)

plt.figure(figsize=(8, 4.5))
plt.plot(list(ks), inercias, 'o-', color='#3b82f6', lw=2)
plt.xlabel('Número de clústeres (k)')
plt.ylabel('Inercia')
plt.title('Método del Codo — ¿Dónde se aplana la curva?')
plt.xticks(list(ks))
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()

for k, iner in zip(ks, inercias):
    print(f"k={k}:  inercia = {iner:.1f}")
# ── Parte 4: Validar K con el Coeficiente de Silhouette ─────
from sklearn.metrics import silhouette_score

print("Silhouette promedio por valor de k (más alto = mejor):")
mejor_k, mejor_sil = None, -1
for k in range(2, 9):
    km = KMeans(n_clusters=k, init='k-means++', n_init=10, random_state=42)
    etiquetas = km.fit_predict(X_scaled)
    sil = silhouette_score(X_scaled, etiquetas)
    marca = ""
    if sil > mejor_sil:
        mejor_sil, mejor_k, marca = sil, k, "  <- mejor hasta ahora"
    print(f"k={k}:  silhouette = {sil:.3f}{marca}")

print(f"\nMejor k según Silhouette: {mejor_k}  (silhouette = {mejor_sil:.3f})")
# ── Parte 5: Interpretar los Clústeres ──────────────────────
# El algoritmo encontró los grupos; ahora les damos significado de negocio.
perfil = df.groupby('cluster')[features].mean().round(1)
print("Perfil promedio de cada clúster:")
print(perfil)

# ¿Coinciden los clústeres con el origen real del auto? (validación externa)
print("\nOrigen de los autos en cada clúster:")
print(pd.crosstab(df['cluster'], df['origin']))

# Lectura típica:
#  - un clúster de autos pesados, mucha cilindrada y poco mpg (muscle cars USA)
#  - un clúster de autos ligeros y eficientes (japoneses/europeos)
#  - un clúster intermedio
print("\n💡 K-means descubrió perfiles de autos SIN ver la etiqueta 'origin'.")
