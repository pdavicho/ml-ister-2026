# ── 1. Importar herramientas ──────────────────────────────
from sklearn.datasets import load_iris
from sklearn.tree import DecisionTreeClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, classification_report

# ── 2. Cargar datos ───────────────────────────────────────
data = load_iris()
X, y = data.data, data.target
print('Clases:', data.target_names)   # ['setosa' 'versicolor' 'virginica']
print('Forma del dataset:', X.shape)  # (150, 4)

# ── 3. Dividir: 70% entrenamiento / 30% prueba ───────────
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.30, random_state=42)

# ── 4. Crear modelo y entrenar ────────────────────────────
model = DecisionTreeClassifier(max_depth=3, criterion='gini')
model.fit(X_train, y_train)           # Aquí ocurre el aprendizaje

# ── 5. Evaluar desempeño ──────────────────────────────────
y_pred = model.predict(X_test)
print('Accuracy:', accuracy_score(y_test, y_pred))
print(classification_report(y_test, y_pred))
# ── Visualización del árbol ───────────────────────────────
from sklearn.tree import plot_tree
import matplotlib.pyplot as plt

plt.figure(figsize=(14, 6))
plot_tree(
    model,
    feature_names=data.feature_names,
    class_names=data.target_names,
    filled=True,    # Colores por clase
    rounded=True,
    impurity=True,  # Muestra Gini en cada nodo
    fontsize=10
)
plt.title('Árbol de Decisión — Dataset Iris')
plt.savefig('arbol_iris.png', dpi=150)
plt.show()

# ── Importancia de variables ──────────────────────────────
import pandas as pd
importancias = pd.Series(
    model.feature_importances_,
    index=data.feature_names
).sort_values(ascending=False)
print(importancias)

Fórmulas de Referencia

Índice de Gini
# Gini = 1 - Σ(pᵢ)²
# donde pᵢ = proporción de la clase i en el nodo
#
# Ejemplo: nodo con 7 Aprobados y 3 Rechazados (total = 10)
p_A = 7 / 10   # 0.70
p_B = 3 / 10   # 0.30
gini = 1 - (p_A**2 + p_B**2)
print(f'Gini = {gini:.4f}')   # Gini = 0.4200
# Gini = 0  → nodo puro   |   Gini = 0.5 → máxima mezcla (2 clases)
Entropía de Shannon
# H = -Σ pᵢ × log₂(pᵢ)
# donde pᵢ = proporción de la clase i en el nodo
#
# Ejemplo: mismo nodo — 7 Aprobados y 3 Rechazados
import math
p_A = 7 / 10   # 0.70
p_B = 3 / 10   # 0.30
H = -(p_A * math.log2(p_A) + p_B * math.log2(p_B))
print(f'Entropía = {H:.4f}')  # Entropía = 0.8816
# H = 0  → nodo puro   |   H = 1.0 → máxima mezcla (2 clases)
Ganancia de Información
# IG = Gini(padre) - Σ[(Nₖ/N) × Gini(hijoₖ)]
#
# Ejemplo: 10 ejemplos en el padre (6 Sí + 4 No)
# División por "¿Nublado?" → Izq: 4 ej. (4 Sí) | Der: 6 ej. (2 Sí + 4 No)

def gini(pos, total):
    if total == 0: return 0
    p = pos / total
    return 1 - p**2 - (1 - p)**2

gini_padre    = gini(6, 10)               # 0.480
gini_izq      = gini(4, 4)               # 0.000  ← nodo puro
gini_der      = gini(2, 6)               # 0.444

ig = gini_padre - ((4/10) * gini_izq + (6/10) * gini_der)
print(f'IG = {ig:.4f}')                   # IG = 0.2160
