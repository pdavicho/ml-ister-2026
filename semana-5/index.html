# ── Parte 1: SVM Lineal — Visualización del Margen ────────
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets        import make_blobs
from sklearn.svm             import SVC
from sklearn.preprocessing   import StandardScaler
from sklearn.model_selection import train_test_split

# Datos sintéticos linealmente separables
X, y = make_blobs(n_samples=80, centers=2, random_state=6, cluster_std=0.8)

scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

svm_lin = SVC(kernel='linear', C=1.0)
svm_lin.fit(X_scaled, y)

# Visualización de la frontera de decisión
xx, yy = np.meshgrid(np.linspace(-3, 3, 200), np.linspace(-3, 3, 200))
Z = svm_lin.decision_function(np.c_[xx.ravel(), yy.ravel()]).reshape(xx.shape)

plt.figure(figsize=(7, 5))
plt.contourf(xx, yy, Z, levels=[-1, 0, 1], alpha=0.15, colors=['#ef4444', '#ffffff', '#3b82f6'])
plt.contour(xx, yy, Z, levels=[-1, 0, 1], colors=['#ef4444', '#1e293b', '#3b82f6'],
            linestyles=['--', '-', '--'], linewidths=[1.5, 2, 1.5])
plt.scatter(X_scaled[y==0, 0], X_scaled[y==0, 1], color='#ef4444', s=50, label='Clase 0', zorder=3)
plt.scatter(X_scaled[y==1, 0], X_scaled[y==1, 1], color='#3b82f6', s=50, label='Clase 1', zorder=3)

# Vectores de soporte
sv = svm_lin.support_vectors_
plt.scatter(sv[:, 0], sv[:, 1], s=200, facecolors='none', edgecolors='yellow',
            linewidths=2, label='Vectores de soporte', zorder=4)

plt.title('SVM Lineal — Hiperplano de Margen Máximo')
plt.xlabel('Feature 1 (escalada)')
plt.ylabel('Feature 2 (escalada)')
plt.legend()
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()

print(f"Vectores de soporte: {len(sv)} puntos")
print(f"Margen = 2 / ‖w‖ = {2 / np.linalg.norm(svm_lin.coef_):.4f}")
# ── Parte 2: Efecto del Parámetro C ───────────────────────
# C pequeño → margen ancho, más errores permitidos
# C grande  → margen estrecho, menos errores, riesgo de overfitting

X, y = make_blobs(n_samples=100, centers=2, random_state=42, cluster_std=1.2)
X_sc = StandardScaler().fit_transform(X)

valores_C = [0.01, 0.1, 1.0, 100.0]
fig, axes = plt.subplots(1, 4, figsize=(16, 4))

for ax, C in zip(axes, valores_C):
    clf = SVC(kernel='linear', C=C)
    clf.fit(X_sc, y)

    xx, yy = np.meshgrid(np.linspace(-3.5, 3.5, 150), np.linspace(-3.5, 3.5, 150))
    Z = clf.decision_function(np.c_[xx.ravel(), yy.ravel()]).reshape(xx.shape)

    ax.contourf(xx, yy, Z, levels=[-1, 0, 1], alpha=0.15, colors=['#ef4444', '#ffffff', '#3b82f6'])
    ax.contour(xx, yy, Z, levels=[-1, 0, 1], colors=['#ef4444', '#1e293b', '#3b82f6'],
               linestyles=['--', '-', '--'], linewidths=[1, 2, 1])
    ax.scatter(X_sc[y==0, 0], X_sc[y==0, 1], color='#ef4444', s=20, alpha=0.7)
    ax.scatter(X_sc[y==1, 0], X_sc[y==1, 1], color='#3b82f6', s=20, alpha=0.7)
    sv = clf.support_vectors_
    ax.scatter(sv[:, 0], sv[:, 1], s=120, facecolors='none', edgecolors='yellow', linewidths=1.5)
    margin = 2 / np.linalg.norm(clf.coef_)
    ax.set_title(f'C = {C}\nMargen = {margin:.2f}  | SVs = {len(sv)}', fontsize=9)
    ax.set_xlim(-3.5, 3.5)
    ax.set_ylim(-3.5, 3.5)
    ax.grid(alpha=0.2)

plt.suptitle('Efecto del parámetro C en SVM Lineal', fontsize=11)
plt.tight_layout()
plt.show()
# ── Parte 3: Kernel RBF — Datos NO linealmente separables ──
from sklearn.datasets import make_circles, make_moons

fig, axes = plt.subplots(2, 3, figsize=(14, 8))

datasets = [
    ('make_moons',   *make_moons(n_samples=150, noise=0.15, random_state=42)),
    ('make_circles', *make_circles(n_samples=150, noise=0.1, factor=0.4, random_state=42)),
]

kernels_C = [('linear', 1), ('rbf', 1), ('rbf', 10)]

for row, (nombre, X_d, y_d) in enumerate(datasets):
    X_s = StandardScaler().fit_transform(X_d)
    for col, (kern, C_val) in enumerate(kernels_C):
        ax = axes[row][col]
        clf = SVC(kernel=kern, C=C_val, gamma='scale')
        clf.fit(X_s, y_d)

        xx, yy = np.meshgrid(np.linspace(-2.5, 2.5, 150), np.linspace(-2.5, 2.5, 150))
        Z = clf.predict(np.c_[xx.ravel(), yy.ravel()]).reshape(xx.shape)
        ax.contourf(xx, yy, Z, alpha=0.25, cmap='coolwarm')
        ax.scatter(X_s[y_d==0, 0], X_s[y_d==0, 1], color='#ef4444', s=20, alpha=0.8)
        ax.scatter(X_s[y_d==1, 0], X_s[y_d==1, 1], color='#3b82f6', s=20, alpha=0.8)
        acc = clf.score(X_s, y_d)
        ax.set_title(f'{nombre}\nkernel={kern} | C={C_val} | acc={acc:.2f}', fontsize=8)
        ax.grid(alpha=0.2)

plt.suptitle('SVM: Kernel Lineal vs RBF en Datos No Lineales', fontsize=11)
plt.tight_layout()
plt.show()

print("Observación: el kernel RBF puede aprender fronteras circulares y curvas")
print("que el kernel lineal NO puede capturar.")
# ── Parte 4: SVM + GridSearchCV — Pipeline Completo ────────
from sklearn.model_selection import GridSearchCV, cross_val_score
from sklearn.pipeline        import Pipeline

# Datos: make_moons (no lineales)
X, y = make_moons(n_samples=300, noise=0.2, random_state=42)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.30, random_state=42)

# Pipeline: escalado + SVM (previene data leakage)
pipe = Pipeline([
    ('scaler', StandardScaler()),
    ('svm',    SVC())
])

param_grid = {
    'svm__kernel': ['linear', 'rbf'],
    'svm__C':      [0.1, 1, 10, 100],
    'svm__gamma':  ['scale', 0.1, 0.01],   # 'scale' ignora gamma para kernel lineal
}

grid = GridSearchCV(pipe, param_grid, cv=5, scoring='accuracy', n_jobs=-1, verbose=0)
grid.fit(X_train, y_train)

print("── GridSearchCV — Mejores hiperparámetros ────────────────────")
print(f"Mejores params: {grid.best_params_}")
print(f"CV-5 accuracy:  {grid.best_score_:.4f}")

y_pred = grid.predict(X_test)
from sklearn.metrics import classification_report
print("\n── Test set ───────────────────────────────────────────────────")
print(classification_report(y_test, y_pred, target_names=['Clase 0', 'Clase 1']))
