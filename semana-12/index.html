# ── Parte 1: Descenso de Gradiente a Mano (NumPy puro) ──────
# Queremos encontrar el w que minimiza L(w) = (w - 3)^2
# El mínimo real es w=3 (lo sabemos porque la función es simple),
# pero el algoritmo lo "descubre" solo, paso a paso.
import numpy as np

def perdida(w):
    return (w - 3) ** 2

def gradiente(w):
    return 2 * (w - 3)          # derivada de (w-3)^2

w = -1.5                        # punto de partida (al azar)
lr = 0.3                        # tasa de aprendizaje

print(f"{'paso':>4} | {'w':>8} | {'L(w)':>8}")
for paso in range(10):
    print(f"{paso:>4} | {w:8.4f} | {perdida(w):8.4f}")
    w = w - lr * gradiente(w)   # la regla de oro del descenso de gradiente

print(f"\nValor final de w: {w:.4f}  (el mínimo real es w=3)")
# ── Parte 2: Visualizar la "Montaña" y el Camino ────────────
import matplotlib.pyplot as plt

def gradient_descent(w0, lr, pasos):
    w = w0
    historial = [w]
    for _ in range(pasos):
        w = w - lr * gradiente(w)
        historial.append(w)
    return historial

camino = gradient_descent(w0=-1.5, lr=0.3, pasos=12)

ws = np.linspace(-3, 9, 200)
plt.figure(figsize=(7, 5))
plt.plot(ws, perdida(ws), color='#334155', lw=2, label='L(w) = (w-3)²')
plt.plot(camino, [perdida(w) for w in camino], 'o-', color='#3b82f6', label='camino del descenso')
plt.scatter([3], [0], color='#22c55e', s=100, zorder=5, label='mínimo (w=3)')
plt.xlabel('w'); plt.ylabel('pérdida L(w)'); plt.legend(); plt.grid(alpha=.3)
plt.title('Descenso de gradiente: cada paso baja un poco la pérdida')
plt.show()
# ── Parte 3: El Efecto de la Tasa de Aprendizaje ────────────
tasas = {'muy baja (0.02)': 0.02, 'buena (0.3)': 0.3, 'muy alta (1.05)': 1.05}

fig, axes = plt.subplots(1, 3, figsize=(15, 4))
for ax, (nombre, lr_i) in zip(axes, tasas.items()):
    camino_i = gradient_descent(w0=-1.5, lr=lr_i, pasos=12)
    ax.plot(ws, perdida(ws), color='#334155', lw=1.5)
    ax.plot(camino_i, [perdida(w) for w in camino_i], 'o-', color='#f59e0b', markersize=4)
    ax.set_title(nombre); ax.set_ylim(0, 40); ax.grid(alpha=.3)

plt.tight_layout()
plt.show()

# Con lr muy alta el "camino" rebota cada vez más lejos del mínimo: diverge.
# Con lr muy baja, en 12 pasos apenas se movió.
# ── Parte 4: Backpropagation a Mano — una Mini-Red para XOR ──
# Red: 2 entradas -> 2 neuronas ocultas -> 1 salida, todas con sigmoide.
# Esta vez SÍ vamos a entrenar los pesos nosotros mismos, sin sklearn.
np.random.seed(42)

def sigmoide(z):       return 1 / (1 + np.exp(-z))
def sigmoide_deriv(a):  return a * (1 - a)      # derivada en función de la SALIDA de la sigmoide

X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([[0], [1], [1], [0]])              # XOR

# Pesos iniciales al azar (pequeños)
W1 = np.random.randn(2, 2) * 0.5   # entradas -> capa oculta (2x2)
b1 = np.zeros((1, 2))
W2 = np.random.randn(2, 1) * 0.5   # capa oculta -> salida (2x1)
b2 = np.zeros((1, 1))

lr = 0.5
historial_loss = []

for epoca in range(5000):
    # ---- FORWARD ----
    z1 = X @ W1 + b1
    a1 = sigmoide(z1)              # activación de la capa oculta
    z2 = a1 @ W2 + b2
    a2 = sigmoide(z2)              # predicción final

    error = a2 - y
    loss = np.mean(error ** 2)
    historial_loss.append(loss)

    # ---- BACKWARD (regla de la cadena, capa por capa) ----
    d2 = error * sigmoide_deriv(a2)            # cuánto "culpa" tiene la salida
    dW2 = a1.T @ d2
    db2 = d2.sum(axis=0, keepdims=True)

    d1 = (d2 @ W2.T) * sigmoide_deriv(a1)      # el error se reparte hacia la capa oculta
    dW1 = X.T @ d1
    db1 = d1.sum(axis=0, keepdims=True)

    # ---- ACTUALIZAR PESOS (descenso de gradiente) ----
    W2 -= lr * dW2; b2 -= lr * db2
    W1 -= lr * dW1; b1 -= lr * db1

print("Predicciones finales (XOR):")
for xi, yi in zip(X, y):
    pred = sigmoide(sigmoide(xi @ W1 + b1) @ W2 + b2)[0][0]
    print(f"  XOR{tuple(xi)} esperado={yi[0]}  predicho={pred:.3f}")

print(f"\nPérdida inicial: {historial_loss[0]:.4f}")
print(f"Pérdida final  : {historial_loss[-1]:.4f}")

plt.figure(figsize=(7, 4))
plt.plot(historial_loss, color='#3b82f6')
plt.xlabel('época'); plt.ylabel('pérdida (MSE)')
plt.title('Backpropagation a mano: la red aprende XOR')
plt.grid(alpha=.3); plt.show()

# La semana pasada dijimos que un perceptrón simple NO puede resolver XOR.
# Hoy, con backpropagation, entrenamos NOSOTROS los pesos de una red con
# capa oculta — y sí lo resuelve.
# ── Parte 5: Lo Mismo, pero con sklearn (por dentro hace lo mismo) ──
from sklearn.neural_network import MLPClassifier

mlp = MLPClassifier(hidden_layer_sizes=(2,), activation='logistic',
                     solver='sgd', learning_rate_init=0.5,
                     max_iter=5000, random_state=42)
mlp.fit(X, y.ravel())

print("Predicciones de MLPClassifier:")
for xi, yi in zip(X, y.ravel()):
    print(f"  XOR{tuple(xi)} esperado={yi}  predicho={mlp.predict([xi])[0]}")

plt.figure(figsize=(7, 4))
plt.plot(mlp.loss_curve_, color='#22c55e')
plt.xlabel('época'); plt.ylabel('pérdida')
plt.title('mlp.loss_curve_ — el mismo descenso de gradiente, calculado por sklearn')
plt.grid(alpha=.3); plt.show()

# Todo lo que programaste a mano en la Parte 4 (forward, backward, actualizar
# pesos) es exactamente lo que hace sklearn "por dentro" cuando llamas .fit()
