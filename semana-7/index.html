# ── Parte 1: Exploración del Dataset Digits ─────────────────
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import load_digits

digits = load_digits()
print(f"Imágenes: {digits.images.shape}  → {digits.data.shape[0]} muestras, {digits.data.shape[1]} features")
print(f"Clases: {digits.target_names}")
print(f"Muestras por clase: {np.bincount(digits.target)}")

# Visualizar algunos dígitos
fig, axes = plt.subplots(3, 10, figsize=(14, 4.5))
for i, ax in enumerate(axes.flat):
    ax.imshow(digits.images[i], cmap='gray_r')
    ax.set_title(str(digits.target[i]), fontsize=9)
    ax.axis('off')
plt.suptitle('Dataset Digits — Primeros 30 ejemplos (8×8 píxeles)', fontsize=11)
plt.tight_layout()
plt.show()
# ── Parte 2: Pipeline SVM con GridSearchCV en Digits ─────────
from sklearn.model_selection import train_test_split, GridSearchCV, cross_val_score
from sklearn.preprocessing   import StandardScaler
from sklearn.svm             import SVC
from sklearn.pipeline        import Pipeline
from sklearn.metrics         import classification_report, ConfusionMatrixDisplay
import time

X = digits.data
y = digits.target

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.25, random_state=42, stratify=y)

# Pipeline: escalar + SVM
pipe_svm = Pipeline([
    ('scaler', StandardScaler()),
    ('svm',    SVC(probability=True))
])

param_grid_svm = {
    'svm__kernel': ['rbf', 'linear'],
    'svm__C':      [0.1, 1, 10],
    'svm__gamma':  ['scale', 0.01],
}

t0 = time.time()
grid_svm = GridSearchCV(pipe_svm, param_grid_svm, cv=5, scoring='accuracy', n_jobs=-1)
grid_svm.fit(X_train, y_train)
t1 = time.time()

print("── SVM GridSearchCV ──────────────────────────────────────────")
print(f"Mejores params: {grid_svm.best_params_}")
print(f"CV-5 accuracy:  {grid_svm.best_score_:.4f}")
print(f"Tiempo búsqueda: {t1-t0:.1f}s")

y_pred_svm = grid_svm.predict(X_test)
acc_svm    = grid_svm.score(X_test, y_test)
print(f"\nTest accuracy: {acc_svm:.4f}")
print(classification_report(y_test, y_pred_svm))
# ── Parte 3: Pipeline K-NN con GridSearchCV en Digits ────────
from sklearn.neighbors import KNeighborsClassifier

pipe_knn = Pipeline([
    ('scaler', StandardScaler()),
    ('knn',    KNeighborsClassifier())
])

param_grid_knn = {
    'knn__n_neighbors': [3, 5, 7, 11],
    'knn__weights':     ['uniform', 'distance'],
    'knn__metric':      ['euclidean', 'manhattan'],
}

t0 = time.time()
grid_knn = GridSearchCV(pipe_knn, param_grid_knn, cv=5, scoring='accuracy', n_jobs=-1)
grid_knn.fit(X_train, y_train)
t1 = time.time()

print("── K-NN GridSearchCV ─────────────────────────────────────────")
print(f"Mejores params: {grid_knn.best_params_}")
print(f"CV-5 accuracy:  {grid_knn.best_score_:.4f}")
print(f"Tiempo búsqueda: {t1-t0:.1f}s")

y_pred_knn = grid_knn.predict(X_test)
acc_knn    = grid_knn.score(X_test, y_test)
print(f"\nTest accuracy: {acc_knn:.4f}")
print(classification_report(y_test, y_pred_knn))
# ── Parte 4: Comparativa Visual SVM vs K-NN ─────────────────
from sklearn.model_selection import learning_curve

fig, axes = plt.subplots(1, 2, figsize=(13, 5))

# Matrices de confusión
ConfusionMatrixDisplay.from_predictions(
    y_test, y_pred_svm, ax=axes[0], colorbar=False, cmap='Blues')
axes[0].set_title(f'SVM (RBF) — Test acc = {acc_svm:.4f}', fontsize=10)

ConfusionMatrixDisplay.from_predictions(
    y_test, y_pred_knn, ax=axes[1], colorbar=False, cmap='Greens')
axes[1].set_title(f'K-NN (k=opt) — Test acc = {acc_knn:.4f}', fontsize=10)

plt.suptitle('Matrices de Confusión — SVM vs K-NN en Dígitos', fontsize=11)
plt.tight_layout()
plt.show()

# Tabla resumen
print("\n══ TABLA COMPARATIVA ════════════════════════════════════════")
print(f"{'Métrica':<25} {'SVM':>10} {'K-NN':>10}")
print("─" * 47)
print(f"{'Test Accuracy':<25} {acc_svm:>10.4f} {acc_knn:>10.4f}")
print(f"{'CV-5 Accuracy':<25} {grid_svm.best_score_:>10.4f} {grid_knn.best_score_:>10.4f}")
ganador = 'SVM' if acc_svm > acc_knn else 'K-NN'
print(f"\n→ Ganador en este dataset: {ganador}")
# ── Parte 5: Curva de Aprendizaje — SVM vs K-NN ─────────────
from sklearn.model_selection import learning_curve

best_svm = grid_svm.best_estimator_
best_knn = grid_knn.best_estimator_

fig, axes = plt.subplots(1, 2, figsize=(13, 4.5))

for ax, modelo, titulo, color in zip(
        axes,
        [best_svm, best_knn],
        ['SVM (Mejor Pipeline)', 'K-NN (Mejor Pipeline)'],
        ['#3b82f6', '#10b981']):

    train_sizes, train_sc, val_sc = learning_curve(
        modelo, X, y,
        train_sizes=np.linspace(0.1, 1.0, 10),
        cv=5, scoring='accuracy', n_jobs=-1
    )
    t_mean = train_sc.mean(axis=1)
    t_std  = train_sc.std(axis=1)
    v_mean = val_sc.mean(axis=1)
    v_std  = val_sc.std(axis=1)

    ax.plot(train_sizes, t_mean, 'o-', color=color, lw=2, label='Train')
    ax.fill_between(train_sizes, t_mean-t_std, t_mean+t_std, alpha=0.15, color=color)
    ax.plot(train_sizes, v_mean, 'o--', color='#f59e0b', lw=2, label='Validación (CV-5)')
    ax.fill_between(train_sizes, v_mean-v_std, v_mean+v_std, alpha=0.15, color='#f59e0b')
    ax.set_xlabel('Muestras de entrenamiento')
    ax.set_ylabel('Accuracy')
    ax.set_title(titulo)
    ax.set_ylim(0.8, 1.02)
    ax.legend()
    ax.grid(alpha=0.3)

plt.suptitle('Curvas de Aprendizaje — SVM vs K-NN en Digits', fontsize=11)
plt.tight_layout()
plt.show()
