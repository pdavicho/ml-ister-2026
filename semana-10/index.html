# ── Parte 1: El Problema — Segmentar las Viviendas de California ──
# Pregunta de negocio: ¿qué TIPOS de zonas residenciales existen en
# California y dónde están? (para una inmobiliaria o un municipio)
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.datasets import fetch_california_housing

housing = fetch_california_housing(as_frame=True)
df = housing.frame.sample(2000, random_state=42).reset_index(drop=True)

print(f"Zonas censales en la muestra: {df.shape[0]}")
print(df[['MedInc', 'HouseAge', 'AveRooms', 'Latitude', 'Longitude']].describe().round(2))

# MedInc   = ingreso mediano de la zona (en decenas de miles de USD)
# HouseAge = antigüedad mediana de las casas (años)
# AveRooms = promedio de habitaciones por vivienda
# Latitude / Longitude = ubicación geográfica
# ── Parte 2: Escalar y Elegir K (codo + silhouette) ─────────
from sklearn.preprocessing import StandardScaler
from sklearn.cluster       import KMeans
from sklearn.metrics       import silhouette_score

features = ['MedInc', 'HouseAge', 'AveRooms', 'Latitude', 'Longitude']
X_scaled = StandardScaler().fit_transform(df[features])

inercias, silhouettes, rango_k = [], [], range(2, 9)
for k in rango_k:
    km  = KMeans(n_clusters=k, init='k-means++', n_init=10, random_state=42)
    lab = km.fit_predict(X_scaled)
    inercias.append(km.inertia_)
    silhouettes.append(silhouette_score(X_scaled, lab))

fig, axes = plt.subplots(1, 2, figsize=(13, 4.5))
axes[0].plot(list(rango_k), inercias, 'o-', color='#3b82f6')
axes[0].set_xlabel('k'); axes[0].set_ylabel('Inercia'); axes[0].set_title('Método del codo')
axes[1].plot(list(rango_k), silhouettes, 'o-', color='#22c55e')
axes[1].set_xlabel('k'); axes[1].set_ylabel('Silhouette'); axes[1].set_title('Silhouette (más alto = mejor)')
for ax in axes: ax.grid(alpha=0.3)
plt.tight_layout()
plt.show()

# El codo y el pico de silhouette sugieren el K de trabajo
# ── Parte 3: K-means Final + el "Mapa" de California ────────
K = 4    # elegido con el codo y silhouette de la Parte 2
km = KMeans(n_clusters=K, init='k-means++', n_init=10, random_state=42)
df['cluster'] = km.fit_predict(X_scaled)

# Al graficar Longitude vs Latitude, los puntos DIBUJAN California
plt.figure(figsize=(7.5, 7))
sc = plt.scatter(df['Longitude'], df['Latitude'], c=df['cluster'],
                 cmap='viridis', s=22, edgecolor='k', linewidth=0.2)
plt.xlabel('Longitud'); plt.ylabel('Latitud')
plt.title(f'Segmentos de zonas residenciales (K-means, k={K})')
plt.colorbar(sc, label='Clúster')
plt.tight_layout()
plt.show()

# Perfilado: el promedio de cada feature por clúster
perfil = df.groupby('cluster')[features].mean().round(2)
perfil['n_zonas'] = df['cluster'].value_counts().sort_index()
print(perfil)
# Lee cada fila y bautiza el segmento:
# ¿ingreso alto o bajo? ¿casas nuevas o viejas? ¿norte o sur? ¿costa o interior?
# ── Parte 4: Evaluar con las Tres Métricas Internas ─────────
from sklearn.metrics import davies_bouldin_score, calinski_harabasz_score

# Fórmula silhouette para UN punto (repaso, ejecutable):
a = 1.2                        # dist. media a los de MI clúster (cohesión)
b = 3.4                        # dist. media al clúster vecino más cercano
s = (b - a) / max(a, b)        # s = 0.647 → bien agrupado
print(f"silhouette del punto = {s:.3f}  (entre -1 y 1, más alto = mejor)")

# Las tres métricas para cada K — ¿coinciden en el mejor?
print(f"\n{'k':>3} {'silhouette':>11} {'davies-bouldin':>15} {'calinski-h':>12}")
for k in range(2, 7):
    lab = KMeans(n_clusters=k, init='k-means++', n_init=10, random_state=42).fit_predict(X_scaled)
    sil = silhouette_score(X_scaled, lab)          # más alto  = mejor
    db  = davies_bouldin_score(X_scaled, lab)      # más BAJO  = mejor
    ch  = calinski_harabasz_score(X_scaled, lab)   # más alto  = mejor
    print(f"{k:>3} {sil:>11.3f} {db:>15.3f} {ch:>12.1f}")

# Si las tres apuntan al mismo k, la decisión es sólida.
# Si se contradicen, gana el k más interpretable para el negocio.
# ── Parte 5: DBScan — ¿Hay Zonas Atípicas? ──────────────────
from sklearn.cluster   import DBSCAN
from sklearn.neighbors import NearestNeighbors

# eps con el gráfico de k-distancias (repaso S9)
min_samples = 6                      # regla: features + 1
nn = NearestNeighbors(n_neighbors=min_samples).fit(X_scaled)
dist, _ = nn.kneighbors(X_scaled)
k_dist = np.sort(dist[:, -1])

plt.figure(figsize=(8, 4))
plt.plot(k_dist, color='#3b82f6', lw=2)
plt.xlabel('Puntos ordenados'); plt.ylabel(f'Distancia al vecino {min_samples}')
plt.title('k-distancias: el codo sugiere eps'); plt.grid(alpha=0.3)
plt.tight_layout(); plt.show()

db = DBSCAN(eps=0.8, min_samples=min_samples)     # eps leído del codo
df['cluster_db'] = db.fit_predict(X_scaled)
es_outlier = df['cluster_db'] == -1

n_clusters = df['cluster_db'].nunique() - (1 if es_outlier.any() else 0)
print(f"DBScan encontró {n_clusters} clústeres y {es_outlier.sum()} zonas atípicas")

plt.figure(figsize=(7.5, 7))
plt.scatter(df.loc[~es_outlier, 'Longitude'], df.loc[~es_outlier, 'Latitude'],
            c=df.loc[~es_outlier, 'cluster_db'], cmap='viridis', s=22, edgecolor='k', linewidth=0.2)
plt.scatter(df.loc[es_outlier, 'Longitude'], df.loc[es_outlier, 'Latitude'],
            c='red', marker='x', s=60, label='Zona atípica (-1)')
plt.xlabel('Longitud'); plt.ylabel('Latitud')
plt.title('DBScan — zonas que no encajan en ningún patrón')
plt.legend(); plt.tight_layout(); plt.show()

# Inspecciona qué hace raras a esas zonas (p.ej. AveRooms altísimo)
print(df[es_outlier][features].describe().round(1))
# ── Parte 6: Del Clúster a la Decisión ──────────────────────
# El entregable de un proyecto de clustering NO es el scatter:
# es una tabla de segmentos con nombre y una acción por segmento.

conclusion = pd.DataFrame({
    'segmento': ['Ej: Costa premium', 'Ej: Interior económico', '...', '...'],
    'perfil':   ['Ingreso alto, costa sur', 'Ingreso bajo, casas antiguas', '...', '...'],
    'accion':   ['Proyectos de lujo', 'Vivienda social / renovación', '...', '...'],
})
print(conclusion.to_string(index=False))
