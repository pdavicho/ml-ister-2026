# ── Parte 1: Cómo Ve una Computadora una Imagen ──────────────
# Toda imagen es, para la computadora, una matriz de números.
!wget -q https://raw.githubusercontent.com/opencv/opencv/master/samples/data/lena.jpg -O foto.jpg

import cv2
import numpy as np
import matplotlib.pyplot as plt

imagen = cv2.imread('foto.jpg')
print("Forma de la imagen (alto, ancho, canales):", imagen.shape)
print("Tipo de dato:", imagen.dtype)
print("Valor del píxel en (100, 100):", imagen[100, 100])   # [B, G, R]

plt.imshow(imagen)
plt.axis('off')
plt.title("cv2.imread — formato BGR (colores raros a propósito)")
plt.show()
# ── Parte 1b: La Trampa del Orden de Colores (BGR vs RGB) ────
# OpenCV guarda los canales como Azul-Verde-Rojo, no Rojo-Verde-Azul.
# Hay que reordenarlos ANTES de mostrar la imagen con matplotlib.
imagen_rgb = cv2.cvtColor(imagen, cv2.COLOR_BGR2RGB)

plt.imshow(imagen_rgb)
plt.axis('off')
plt.title("Después de cv2.cvtColor(..., COLOR_BGR2RGB)")
plt.show()

# Regla práctica de hoy: convertir a RGB solo para MOSTRAR con matplotlib.
# Para procesar o detectar, muchas funciones de OpenCV siguen esperando BGR.
# ── Parte 2: Preprocesamiento de Imágenes ────────────────────
gris    = cv2.cvtColor(imagen, cv2.COLOR_BGR2GRAY)
print("Forma en escala de grises:", gris.shape)   # ya no tiene el 3 de los canales

pequena = cv2.resize(imagen_rgb, (128, 128))
borrosa = cv2.GaussianBlur(imagen_rgb, (15, 15), 0)
bordes  = cv2.Canny(gris, threshold1=100, threshold2=200)

fig, axes = plt.subplots(1, 4, figsize=(16, 4))
axes[0].imshow(gris, cmap='gray');    axes[0].set_title('Escala de grises'); axes[0].axis('off')
axes[1].imshow(pequena);              axes[1].set_title('Resize 128×128');   axes[1].axis('off')
axes[2].imshow(borrosa);              axes[2].set_title('Gaussian Blur');    axes[2].axis('off')
axes[3].imshow(bordes, cmap='gray');  axes[3].set_title('Canny (bordes)');   axes[3].axis('off')
plt.tight_layout()
plt.show()

# Cada una de estas transformaciones es un paso de PREPROCESAMIENTO:
# preparar la imagen antes de dársela a un modelo (reducir tamaño,
# simplificar colores, resaltar formas) para que el modelo entrene
# más rápido y se fije en lo importante.
# ── Parte 3: Clasificación de Imágenes con una CNN (Keras) ───
# Dataset MNIST: dígitos manuscritos 28x28 en escala de grises.
from tensorflow.keras import layers, models
from tensorflow.keras.datasets import mnist

(x_train, y_train), (x_test, y_test) = mnist.load_data()

# Nos quedamos con una parte para que el entrenamiento sea rápido en clase
x_train, y_train = x_train[:4000], y_train[:4000]
x_test,  y_test  = x_test[:1000],  y_test[:1000]

x_train = x_train.reshape(-1, 28, 28, 1) / 255.0
x_test  = x_test.reshape(-1, 28, 28, 1) / 255.0

plt.figure(figsize=(8, 2))
for i in range(6):
    plt.subplot(1, 6, i + 1)
    plt.imshow(x_train[i].reshape(28, 28), cmap='gray')
    plt.title(y_train[i]); plt.axis('off')
plt.show()
# ── Parte 3b: Construir y Entrenar la CNN ────────────────────
modelo = models.Sequential([
    layers.Conv2D(16, (3, 3), activation='relu', input_shape=(28, 28, 1)),
    layers.MaxPooling2D((2, 2)),
    layers.Conv2D(32, (3, 3), activation='relu'),
    layers.MaxPooling2D((2, 2)),
    layers.Flatten(),
    layers.Dense(32, activation='relu'),
    layers.Dense(10, activation='softmax')
])
modelo.summary()

modelo.compile(optimizer='adam',
               loss='sparse_categorical_crossentropy',
               metrics=['accuracy'])

historial = modelo.fit(x_train, y_train, epochs=5,
                        validation_data=(x_test, y_test))

plt.plot(historial.history['accuracy'], label='entrenamiento')
plt.plot(historial.history['val_accuracy'], label='validación')
plt.xlabel('época'); plt.ylabel('accuracy'); plt.legend()
plt.title('CNN aprendiendo a reconocer dígitos')
plt.grid(alpha=.3); plt.show()

# Conv2D = filtros que se deslizan buscando patrones (bordes, curvas, trazos).
# MaxPooling2D = se queda con lo más fuerte de cada zona y reduce el tamaño.
# Flatten + Dense = igual que en la Semana 11/12, decide la clase final.
# ── Parte 4: Aplicaciones Reales — Detección de Rostros ──────
# Un clasificador clásico de OpenCV (Haar Cascade), ya entrenado,
# que encuentra rostros sin que nosotros entrenemos nada.
face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml')

caras = face_cascade.detectMultiScale(gris, scaleFactor=1.1, minNeighbors=5)

resultado = imagen_rgb.copy()
for (x, y, w, h) in caras:
    cv2.rectangle(resultado, (x, y), (x + w, y + h), (0, 255, 0), 3)

plt.imshow(resultado)
plt.axis('off')
plt.title(f"{len(caras)} rostro(s) detectado(s)")
plt.show()

# Esto es SOLO detección: encontrar dónde está la cara.
