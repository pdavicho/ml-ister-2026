# ── Parte 1: K-NN — Visualización del algoritmo ─────────────
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets        import make_blobs
from sklearn.neighbors       import KNeighborsClassifier
from sklearn.preprocessing   import StandardScaler
from sklearn.model_selection import train_test_split

# Datos sintéticos en 2D para visualizar K-NN
X, y = make_blobs(n_samples=120, centers=3, random_state=42, cluster_std=1.0)

scaler  = StandardScaler()
X_sc    = scaler.fit_transform(X)

# Probar K = 1, 5, 15
fig, axes = plt.subplots(1, 3, figsize=(15, 4))

for ax, k in zip(axes, [1, 5, 15]):
    knn = KNeighborsClassifier(n_neighbors=k)
    knn.fit(X_sc, y)

    xx, yy = np.meshgrid(np.linspace(-3, 3, 200), np.linspace(-3, 3, 200))
    Z = knn.predict(np.c_[xx.ravel(), yy.ravel()]).reshape(xx.shape)

    ax.contourf(xx, yy, Z, alpha=0.25, cmap='Set1')
    ax.scatter(X_sc[y==0, 0], X_sc[y==0, 1], color='#ef4444', s=30, alpha=0.8, label='Clase 0')
    ax.scatter(X_sc[y==1, 0], X_sc[y==1, 1], color='#3b82f6', s=30, alpha=0.8, label='Clase 1')
    ax.scatter(X_sc[y==2, 0], X_sc[y==2, 1], color='#10b981', s=30, alpha=0.8, label='Clase 2')
    acc = knn.score(X_sc, y)
    ax.set_title(f'K = {k}  |  acc (train) = {acc:.3f}', fontsize=10)
    ax.grid(alpha=0.2)

plt.suptitle('K-NN — Efecto de K en la Frontera de Decisión', fontsize=11)
plt.tight_layout()
plt.show()
# ── Parte 2: Métricas de Distancia ──────────────────────────
import math

# Distancia Euclídea, Manhattan y Minkowski en Python
def distancia_euclidea(a, b):
    return math.sqrt(sum((x - y)**2 for x, y in zip(a, b)))

def distancia_manhattan(a, b):
    return sum(abs(x - y) for x, y in zip(a, b))

def distancia_minkowski(a, b, p):
    return sum(abs(x - y)**p for x, y in zip(a, b))**(1/p)

# Ejemplo
punto_A = [2.0, 3.0, 5.0]
punto_B = [4.0, 1.0, 3.0]

de = distancia_euclidea(punto_A, punto_B)
dm = distancia_manhattan(punto_A, punto_B)
dm2 = distancia_minkowski(punto_A, punto_B, p=2)

print("── Comparativa de distancias ─────────────────────────────")
print(f"Punto A: {punto_A}")
print(f"Punto B: {punto_B}")
print(f"Distancia Euclídea   (p=2): {de:.4f}")
print(f"Distancia Manhattan  (p=1): {dm:.4f}")
print(f"Distancia Minkowski  (p=2): {dm2:.4f}  ← igual que Euclídea")
# ── Parte 3: K-NN en Titanic — Dataset Real ─────────────────
import pandas as pd
import seaborn as sns
from sklearn.neighbors import KNeighborsClassifier
from sklearn.preprocessing   import StandardScaler
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.metrics         import classification_report

# Cargar y preparar Titanic
df = sns.load_dataset('titanic')
features = ['pclass', 'sex', 'age', 'fare']
df_clean = df[features + ['survived']].dropna().copy()
df_clean['sex'] = (df_clean['sex'] == 'male').astype(int)

X = df_clean[features].values
y = df_clean['survived'].values

print(f"Dataset Titanic limpio: {X.shape[0]} pasajeros, {X.shape[1]} features")

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.30, random_state=42, stratify=y)

# ── Parte 4: Curva de K óptimo ──────────────────────────────
# ¿Cuál es el mejor K? Lo buscamos con cross-validation

from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import cross_val_score

X_sc_all = StandardScaler().fit_transform(X)

k_values = range(1, 31)
cv_scores = []

for k in k_values:
    knn = KNeighborsClassifier(n_neighbors=k, weights='distance')
    scores = cross_val_score(knn, X_sc_all, y, cv=10, scoring='accuracy')
    cv_scores.append(scores.mean())

mejor_k = k_values[cv_scores.index(max(cv_scores))]
print(f"Mejor K encontrado: {mejor_k}  | CV-10 accuracy: {max(cv_scores):.4f}")

plt.figure(figsize=(9, 4))
plt.plot(list(k_values), cv_scores, 'o-', color='#3b82f6', lw=2, markersize=5)
plt.axvline(mejor_k, color='#f59e0b', linestyle='--', lw=1.5, label=f'K óptimo = {mejor_k}')
plt.xlabel('Valor de K')
plt.ylabel('CV-10 Accuracy (media)')
plt.title('Curva de K óptimo — K-NN en Titanic (con normalización)')
plt.legend()
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()
