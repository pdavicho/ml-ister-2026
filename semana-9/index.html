# ── Parte 1: Tres Geometrías para Poner a Prueba el Clustering ──
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets      import make_moons, make_circles, make_blobs
from sklearn.preprocessing import StandardScaler

np.random.seed(42)
datasets = {
    'Blobs (esférico)':      make_blobs(n_samples=300, centers=3, cluster_std=1.0, random_state=42),
    'Moons (alargado)':      make_moons(n_samples=300, noise=0.07, random_state=42),
    'Circles (anidado)':     make_circles(n_samples=300, noise=0.06, factor=0.4, random_state=42),
}

# Escalamos cada dataset (obligatorio para distancias y densidades)
datos_scaled = {}
for nombre, (Xd, yd) in datasets.items():
    datos_scaled[nombre] = StandardScaler().fit_transform(Xd)
    print(f"{nombre}: {Xd.shape[0]} puntos")
# ── Parte 2: K-means en las Tres Geometrías ─────────────────
from sklearn.cluster import KMeans

fig, axes = plt.subplots(1, 3, figsize=(15, 4.5))
for ax, (nombre, Xs) in zip(axes, datos_scaled.items()):
    k = 3 if 'Blobs' in nombre else 2
    km = KMeans(n_clusters=k, init='k-means++', n_init=10, random_state=42)
    etiquetas = km.fit_predict(Xs)
    ax.scatter(Xs[:, 0], Xs[:, 1], c=etiquetas, cmap='viridis', s=25, edgecolor='k', linewidth=0.2)
    ax.scatter(km.cluster_centers_[:, 0], km.cluster_centers_[:, 1],
               c='red', marker='X', s=160)
    ax.set_title(f'K-means (k={k})\n{nombre}', fontsize=10)
    ax.set_xticks([]); ax.set_yticks([])

plt.suptitle('K-means: perfecto en Blobs, FALLA en Moons y Circles', fontsize=11)
plt.tight_layout()
plt.show()

# K-means corta las lunas y los anillos por la mitad: solo entiende grupos redondos
# ── Parte 3: DBScan en las Mismas Geometrías ────────────────
from sklearn.cluster import DBSCAN

# eps ajustado para datos escalados; min_samples=5 (regla para 2D)
eps_por_dataset = {'Blobs (esférico)': 0.4, 'Moons (alargado)': 0.25, 'Circles (anidado)': 0.25}

fig, axes = plt.subplots(1, 3, figsize=(15, 4.5))
for ax, (nombre, Xs) in zip(axes, datos_scaled.items()):
    db = DBSCAN(eps=eps_por_dataset[nombre], min_samples=5)
    etiquetas = db.fit_predict(Xs)
    n_clusters = len(set(etiquetas)) - (1 if -1 in etiquetas else 0)
    n_ruido    = list(etiquetas).count(-1)
    ax.scatter(Xs[:, 0], Xs[:, 1], c=etiquetas, cmap='viridis', s=25, edgecolor='k', linewidth=0.2)
    ax.set_title(f'DBScan — {nombre}\nclusters={n_clusters} · ruido={n_ruido}', fontsize=10)
    ax.set_xticks([]); ax.set_yticks([])

plt.suptitle('DBScan: sigue la FORMA de los datos (los puntos morados = ruido, -1)', fontsize=11)
plt.tight_layout()
plt.show()

# DBScan separa correctamente las lunas y los anillos siguiendo la densidad
# ── Parte 4: Elegir eps con el Gráfico de k-distancias ──────
from sklearn.neighbors import NearestNeighbors

Xs = datos_scaled['Moons (alargado)']
min_samples = 5

# Distancia de cada punto a su k-ésimo vecino más cercano
nn = NearestNeighbors(n_neighbors=min_samples)
nn.fit(Xs)
distancias, _ = nn.kneighbors(Xs)
k_dist = np.sort(distancias[:, -1])    # distancia al 5º vecino, ordenada

plt.figure(figsize=(8, 4.5))
plt.plot(k_dist, color='#3b82f6', lw=2)
plt.xlabel('Puntos ordenados')
plt.ylabel(f'Distancia al vecino {min_samples}')
plt.title('Gráfico de k-distancias — el "codo" sugiere un buen eps')
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()

print("Busca el codo de la curva: esa altura es un buen valor de eps (~0.20-0.30 aquí)")
# ── Parte 5: DBScan como Detector de Outliers ───────────────
# Generamos blobs limpios + puntos de ruido aleatorio inyectado
X_blobs, _ = make_blobs(n_samples=280, centers=3, cluster_std=0.6, random_state=42)
ruido      = np.random.uniform(low=X_blobs.min()-2, high=X_blobs.max()+2, size=(20, 2))
X_total    = np.vstack([X_blobs, ruido])
X_total    = StandardScaler().fit_transform(X_total)

db = DBSCAN(eps=0.3, min_samples=5)
etiquetas = db.fit_predict(X_total)
es_outlier = etiquetas == -1

plt.figure(figsize=(7.5, 5.5))
plt.scatter(X_total[~es_outlier, 0], X_total[~es_outlier, 1],
            c=etiquetas[~es_outlier], cmap='viridis', s=30, edgecolor='k', linewidth=0.2, label='Clústeres')
plt.scatter(X_total[es_outlier, 0], X_total[es_outlier, 1],
            c='red', marker='x', s=80, label='Outliers (-1)')
plt.title('DBScan detecta outliers automáticamente')
plt.legend()
plt.tight_layout()
plt.show()

print(f"Puntos marcados como outliers: {es_outlier.sum()} de {len(X_total)}")
print("→ Útil para detección de fraude, sensores defectuosos, anomalías.")
