# ── Parte 1: La Neurona Artificial con NumPy ────────────────
# Una neurona: entradas (x), pesos (w), sesgo (b) y una suma ponderada.
import numpy as np

x = np.array([1, 0])       # entradas (x1, x2)
w = np.array([1, 1])       # pesos (w1, w2)
b = -1.5                   # sesgo (bias)

z = np.dot(w, x) + b       # suma ponderada: z = w1*x1 + w2*x2 + b
print(f"z = w·x + b = {z}")

def escalon(z):
    return 1 if z >= 0 else 0

y = escalon(z)
print(f"activación escalón: y = {y}")
# Cambia x, w o b y vuelve a ejecutar: así "programa" una neurona a mano
# ── Parte 2: Funciones de Activación ────────────────────────
import matplotlib.pyplot as plt

def escalon_v(z):  return np.where(z >= 0, 1, 0)
def sigmoide(z):   return 1 / (1 + np.exp(-z))
def tanh_f(z):      return np.tanh(z)
def relu(z):        return np.maximum(0, z)

z = np.linspace(-5, 5, 200)
funciones = [('Escalón', escalon_v), ('Sigmoide', sigmoide), ('Tanh', tanh_f), ('ReLU', relu)]

fig, axes = plt.subplots(1, 4, figsize=(16, 3.2))
for ax, (nombre, f) in zip(axes, funciones):
    ax.plot(z, f(z), color='#3b82f6', lw=2)
    ax.set_title(nombre); ax.grid(alpha=.3)
    ax.axhline(0, color='gray', lw=.5); ax.axvline(0, color='gray', lw=.5)
plt.tight_layout()
plt.show()

# Escalón: {0,1}. Sigmoide: (0,1). Tanh: (-1,1). ReLU: [0,∞)
# Todas "deciden" qué tan fuerte dispara la neurona, cada una a su manera.
# ── Parte 3: Un Perceptrón Resuelve AND y OR ────────────────
def perceptron(x1, x2, w1, w2, b):
    z = w1 * x1 + w2 * x2 + b
    return 1 if z >= 0 else 0

print("Compuerta AND (w1=1, w2=1, b=-1.5)")
for x1 in [0, 1]:
    for x2 in [0, 1]:
        print(f"  AND({x1},{x2}) = {perceptron(x1, x2, 1, 1, -1.5)}")

print("\nCompuerta OR (w1=1, w2=1, b=-0.5)")
for x1 in [0, 1]:
    for x2 in [0, 1]:
        print(f"  OR({x1},{x2}) = {perceptron(x1, x2, 1, 1, -0.5)}")

# Los mismos pesos (w1=1, w2=1) con un sesgo distinto cambian la compuerta.
# El sesgo mueve la línea; los pesos la inclinan.
# ── Parte 4: Graficar la Frontera de Decisión ───────────────
def frontera(w1, w2, b, ax, titulo):
    puntos = [(0, 0), (0, 1), (1, 0), (1, 1)]
    for x1, x2 in puntos:
        y = perceptron(x1, x2, w1, w2, b)
        ax.scatter(x1, x2, s=200, c='#22c55e' if y == 1 else '#ef4444', edgecolor='k', zorder=3)
        ax.annotate(f'({x1},{x2})', (x1, x2), textcoords="offset points", xytext=(10, 10))
    xs = np.linspace(-0.5, 1.5, 10)
    ys = -(w1 * xs + b) / w2          # despejando x2 de w1*x1 + w2*x2 + b = 0
    ax.plot(xs, ys, 'b--', lw=2)
    ax.set_xlim(-.5, 1.5); ax.set_ylim(-.5, 1.5); ax.set_title(titulo); ax.grid(alpha=.3)

fig, axes = plt.subplots(1, 2, figsize=(10, 4.5))
frontera(1, 1, -1.5, axes[0], 'AND — una línea SÍ separa')
frontera(1, 1, -0.5, axes[1], 'OR — una línea SÍ separa')
plt.tight_layout()
plt.show()
# ── Parte 5: El Problema XOR — el Perceptrón Falla ──────────
print("Compuerta XOR (intento con w1=1, w2=1, b=-0.5)")
for x1 in [0, 1]:
    for x2 in [0, 1]:
        esperado = x1 ^ x2
        obtenido = perceptron(x1, x2, 1, 1, -0.5)
        marca = "✅" if esperado == obtenido else "❌"
        print(f"  XOR({x1},{x2}) esperado={esperado} obtenido={obtenido} {marca}")

# Prueba a mano otros valores de w1, w2, b: ningún trío logra
# que las 4 combinaciones den el resultado correcto.
# XOR NO es separable con una sola línea recta.
# ── Parte 6: La Solución — una Red con Capa Oculta ──────────
from sklearn.neural_network import MLPClassifier

X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([0, 1, 1, 0])   # XOR

mlp = MLPClassifier(hidden_layer_sizes=(4,), activation='tanh',
                     max_iter=5000, random_state=42)
mlp.fit(X, y)

print("Predicciones de la red (1 capa oculta con 4 neuronas):")
for xi, yi in zip(X, y):
    pred = mlp.predict([xi])[0]
    print(f"  XOR{tuple(xi)} esperado={yi} predicho={pred}")

print(f"\nAccuracy: {mlp.score(X, y):.2f}")
# Con una capa OCULTA, la red combina varias líneas y SÍ resuelve XOR.
# El "cómo" se entrena esa capa (backpropagation) llega en la Semana 13.
